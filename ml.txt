# ML ONLY — ads stays on ~/gratify-venv (do not touch ads)
# ============================================================
# Your Pi OS Trixie has ONLY Python 3.13.
# apt has NO python3.11 package.
# Miniforge has NO 32-bit (armv7l) installer.
#
# Fix: download portable Python 3.11 for Pi armv7hf, then
# install tflite-runtime into a SEPARATE ML venv.
# ============================================================

# 1) Download portable CPython 3.11 (armv7 hard-float = Pi 3A+ 32-bit)
cd ~
curl -L -o python311.tar.gz \
  "https://github.com/astral-sh/python-build-standalone/releases/download/20260807/cpython-3.11.15%2B20260807-armv7-unknown-linux-gnueabihf-install_only.tar.gz"

# 2) Extract (creates ~/python for this tarball layout — we move it)
rm -rf ~/gratify-python311
mkdir -p ~/gratify-python311
tar -xzf python311.tar.gz -C ~/gratify-python311 --strip-components=1
rm -f python311.tar.gz
~/gratify-python311/bin/python3.11 -V
# must show Python 3.11.x

# 3) Create ML-only venv (never touch ~/gratify-venv)
~/gratify-python311/bin/python3.11 -m venv ~/gratify-ml-venv
source ~/gratify-ml-venv/bin/activate
python -V
# must show Python 3.11.x — if 3.13, STOP

# 4) Install ML packages
pip install -U pip setuptools wheel
pip install numpy opencv-python-headless pyserial requests
pip install tflite-runtime==2.14.0

# 5) Verify
python -c "from tflite_runtime.interpreter import Interpreter; print('tflite OK')"
cd ~/ServiceBin
test -f GarbageDetection/main.py || { echo "MISSING GarbageDetection/main.py"; exit 1; }
test -f GarbageDetection/model/rls_mobilenetv3_mish_cbam_float16.tflite || { echo "MISSING model"; exit 1; }
python -c "from GarbageDetection.classifier import predict; import numpy as np; print(predict(np.zeros((480,640,3), dtype=np.uint8)))"

# 6) Point ONLY gratify-garbage at ML venv (ads unchanged)
sudo mkdir -p /etc/systemd/system/gratify-garbage.service.d
sudo tee /etc/systemd/system/gratify-garbage.service.d/override.conf >/dev/null <<EOF
[Service]
ExecStart=
ExecStart=$HOME/gratify-ml-venv/bin/python -m GarbageDetection.main
EOF

sudo systemctl daemon-reload
sudo systemctl restart gratify-garbage
sudo systemctl status gratify-garbage --no-pager
