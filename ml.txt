  /usr/lib/$ARCH_LIB/libavcodec.so.59 \

  /usr/lib/$ARCH_LIB/libavformat.so.59 \
  /usr/lib/$ARCH_LIB/libavformat.so.60 \
  /usr/lib/$ARCH_LIB/libavutil.so.57 \
  /usr/lib/$ARCH_LIB/libswscale.so.6 \

# 2) System tools ML camera needs
sudo apt update
sudo apt install -y ffmpeg libopenblas0 libblas3 liblapack3 libgomp1 || true

# 3) Pull latest ML code (no OpenCV)
cd ~/ServiceBin
git pull

# 4) Fix ML venv packages
source ~/gratify-ml-venv/bin/activate

# 5) Test (no cv2)
python -c "from tflite_runtime.interpreter import Interpreter; print('tflite OK')"

which ffmpeg

# 6) Point ONLY garbage at ML venv (if not already)
sudo mkdir -p /etc/systemd/system/gratify-garbage.service.d
sudo tee /etc/systemd/system/gratify-garbage.service.d/override.conf >/dev/null <<EOF

ExecStart=
ExecStart=$HOME/gratify-ml-venv/bin/python -m GarbageDetection.main

sudo systemctl status gratify-garbage --no-pager
