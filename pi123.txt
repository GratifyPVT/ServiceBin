# Pi 3A+ / Raspberry Pi OS Trixie (Python 3.13, armv7l)
# ============================================================
# WHY THINGS FAIL:
# - pip has NO tflite-runtime for Python 3.13 on Pi
# - apt has NO python3-tflite-runtime on Trixie
# - so "pip install -r requirements.txt" dies and requests never installs
# - ads then crashes: ModuleNotFoundError: No module named 'requests'
#
# FIX: install core packages WITHOUT tflite first (ads+waste work).
#      then install Miniforge + Python 3.11 for ML (tflite works there).
# ============================================================

# ------------------------------------------------------------
# A) FIX ADS + WASTE NOW (use current venv / Python 3.13)
# ------------------------------------------------------------
source ~/gratify-venv/bin/activate
cd ~/ServiceBin

# DO NOT use old requirements.txt if it still lists tflite-runtime
pip install -U pip
pip install requests numpy opencv-python-headless pyserial

python -c "import requests; print('requests OK')"
which mpv || sudo apt install -y mpv
ls AdsManagement/videos/*.mp4

sudo systemctl stop gratify-ads gratify-waste 2>/dev/null
CONDA_PYTHON=$HOME/gratify-venv/bin/python bash deploy/install-services.sh
sudo systemctl restart gratify-ads gratify-waste
sudo systemctl status gratify-ads --no-pager

# manual ads test (Ctrl+C to stop):
# cd ~/ServiceBin && source ~/gratify-venv/bin/activate && python -m AdsManagement.service

# ------------------------------------------------------------
# B) FIX ML / GARBAGE (needs Python 3.11 + tflite-runtime)
#    Python 3.13 on Trixie cannot install tflite — use Miniforge.
# ------------------------------------------------------------
cd ~
ARCH="$(uname -m)"
# armv7l = 32-bit Pi OS (normal for Pi 3A+)
curl -L -o Miniforge3.sh \
  "https://github.com/conda-forge/miniforge/releases/latest/download/Miniforge3-Linux-${ARCH}.sh"

# If armv7l download 404s, try older tag manually, OR flash 64-bit Pi OS and re-run.
bash Miniforge3.sh -b -p $HOME/miniforge3
rm -f Miniforge3.sh
$HOME/miniforge3/bin/conda init bash
source ~/.bashrc

conda create -n mf python=3.11 -y
conda activate mf
python -V
# must show Python 3.11.x

cd ~/ServiceBin
pip install -U pip
pip install requests numpy opencv-python-headless pyserial tflite-runtime

python -c "import requests; print('requests OK')"
python -c "from tflite_runtime.interpreter import Interpreter; print('tflite OK')"
ls GarbageDetection/model/*.tflite

# Point ALL services at conda python 3.11
CONDA_PYTHON=$HOME/miniforge3/envs/mf/bin/python bash deploy/install-services.sh

sudo systemctl restart gratify-update gratify-garbage gratify-ads gratify-waste
sudo systemctl status gratify-garbage gratify-ads gratify-waste --no-pager
