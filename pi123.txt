# Fix ads crash + tflite on Pi OS (Python 3.13 / Pi 3A+)
# Root cause: pip has NO tflite-runtime for py3.13 armv7l,
# so "pip install -r requirements.txt" failed and requests never installed.

# 1) See error
journalctl -u gratify-ads -n 40 --no-pager

# 2) Install tflite from apt (works on Pi OS)
sudo apt update
sudo apt install -y mpv python3-tflite-runtime

# 3) Recreate venv so it can see apt Python packages
deactivate 2>/dev/null
rm -rf ~/gratify-venv
python3 -m venv --system-site-packages ~/gratify-venv
source ~/gratify-venv/bin/activate
pip install -U pip
cd ~/ServiceBin
pip install numpy opencv-python-headless pyserial requests

# 4) Verify
python -c "import requests; print('requests OK')"
python -c "from tflite_runtime.interpreter import Interpreter; print('tflite OK')"
which mpv

# 5) Reinstall systemd services to use this venv
cd ~/ServiceBin
CONDA_PYTHON=$HOME/gratify-venv/bin/python bash deploy/install-services.sh

# 6) Test ads (Ctrl+C to stop)
cd ~/ServiceBin
python -m AdsManagement.service

# 7) Restart
sudo systemctl restart gratify-ads gratify-waste gratify-garbage
sudo systemctl status gratify-ads --no-pager
